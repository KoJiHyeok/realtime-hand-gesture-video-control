왼손·오른손이 반대직접 테스트에서 반대로 나왔고, 샘플 사진 한 장으로 비교하니 원본은 맞고 좌우 반전본은 틀렸습니다. 설계 문서의 "거울 기준" 추정과 달랐습니다.좌우 반전을 빼고, 쓸기 방향 부호를 뒤집었습니다.
쓸기가 거의 안 찍힘로그에서 빠르게 움직일 때 제스처 아님(당시 표시 None)과 손 없음이 끼어듭니다. 웹캠 측정은 28.6 FPS라 속도 문제는 아닙니다.손을 잠깐 놓쳐도 기록을 유지하고, 쓸기는 연속 프레임 없이 바로 인정합니다.

"""메인 루프: 프레임 읽기 → 인식 → 상태 → 명령 → 필터 → 키 입력. 종료는 Ctrl+C.

python main.py         실제 키 입력
python main.py --test  키 없이 제스처 인식만 확인
"""
import sys
import time

import cv2

import config
import controller
import gestures
from check import Checklist
from recognizer import Recognizer
from state import RemoteState


def status_line(mode, hands):
    parts = [f"{config.NAMES[hand.side]}: {config.NAMES[hand.gesture]}" for hand in hands]
    return f"[{config.NAMES[mode]}] " + (" / ".join(parts) if parts else "손 없음")


def main():
    camera = cv2.VideoCapture(config.CAMERA_INDEX)  # OpenCV 는 프레임 읽기에만 쓴다
    if not camera.isOpened():
        raise SystemExit("카메라를 열 수 없습니다.")

    recognizer = Recognizer()
    swipe = gestures.SwipeDetector()
    state = RemoteState()
    last_line = None
    checklist = Checklist() if "--test" in sys.argv else None
    print("시작: 양손 손바닥 1초 / 종료: 양손 주먹 1초 / 프로그램 끝내기: Ctrl+C")

    try:
        while True:
            ok, frame = camera.read()
            if not ok:
                continue

            # BGR → RGB. 좌우 반전은 하지 않는다 (반전하면 왼손/오른손 라벨이 뒤바뀐다)
            rgb = frame[:, :, ::-1].copy()
            now = time.monotonic()
            hands = recognizer.recognize(rgb, int(now * 1000))

            before = state.mode
            state.update_mode(now, hands)
            if checklist and state.mode != before:
                if state.mode == "ACTIVE":
                    checklist.mark("시작")
                elif gestures.both_hands(hands, "Closed_Fist"):  # 손 없음 5초로 꺼진 건 제외
                    checklist.mark("종료")

            line = status_line(state.mode, hands)
            if line != last_line:
                print(line)
                last_line = line

            if state.mode == "ACTIVE":
                command = gestures.get_command(hands, swipe, now)
                if state.should_fire(command, now):
                    key = config.KEYS[command]
                    name = config.NAMES[command]
                    print(f"[{config.NAMES['ACTIVE']}] {name} -> {key} 키")
                    if checklist:
                        checklist.mark(name)
                    else:
                        controller.press(key)
            else:
                swipe.reset()
    except KeyboardInterrupt:
        pass
    finally:
        camera.release()
        recognizer.close()


if __name__ == "__main__":
    main()

MediaPipe를 호출하는 유일한 파일입니다.

"""MediaPipe GestureRecognizer 를 감싼다. 프레임을 넣으면 손 목록이 나온다."""
from collections import namedtuple

import mediapipe as mp
from mediapipe.tasks import python as mp_python
from mediapipe.tasks.python import vision

import config

Hand = namedtuple("Hand", "side gesture score landmarks")


class Recognizer:
    def __init__(self):
        options = vision.GestureRecognizerOptions(
            base_options=mp_python.BaseOptions(model_asset_path=config.MODEL_PATH),
            running_mode=vision.RunningMode.VIDEO,
            num_hands=2,
            min_hand_detection_confidence=config.MIN_CONFIDENCE,
            min_tracking_confidence=config.MIN_CONFIDENCE,
        )
        self.recognizer = vision.GestureRecognizer.create_from_options(options)

    def recognize(self, rgb_frame, timestamp_ms):
        image = mp.Image(image_format=mp.ImageFormat.SRGB, data=rgb_frame)
        # VIDEO 모드는 프레임 순서를 알아야 해서 시각(timestamp_ms)을 함께 넘긴다
        result = self.recognizer.recognize_for_video(image, timestamp_ms)

        hands = []
        for i, landmarks in enumerate(result.hand_landmarks):
            best = result.gestures[i][0]
            gesture = best.category_name if best.score >= config.MIN_SCORE else "None"
            side = result.handedness[i][0].category_name
            hands.append(Hand(side, gesture, best.score, landmarks))
        return hands

    def close(self):
        self.recognizer.close()

INDEX_TIP = 8
MIDDLE_KNUCKLE = 9
OTHER_FINGERS = [(12, 10), (16, 14), (20, 18)]  # (손끝, 관절) 중지·약지·새끼


def distance(a, b):
    return math.hypot(a.x - b.x, a.y - b.y)


def hand_size(landmarks):
    """손목~중지 뿌리 거리. 카메라와의 거리가 달라져도 비율로 비교할 수 있게 해 준다."""
    return distance(landmarks[WRIST], landmarks[MIDDLE_KNUCKLE])


def is_finger_open(landmarks, tip, joint):
    """손끝이 관절보다 손목에서 멀면 펴진 것."""
    wrist = landmarks[WRIST]
    return distance(landmarks[tip], wrist) > distance(landmarks[joint], wrist)


def is_ok(landmarks):
    touching = (distance(landmarks[THUMB_TIP], landmarks[INDEX_TIP])
                < hand_size(landmarks) * config.OK_RATIO)
    others_open = all(is_finger_open(landmarks, tip, joint) for tip, joint in OTHER_FINGERS)
    return touching and others_open


def both_hands(hands, gesture):
    return len(hands) == 2 and all(hand.gesture == gesture for hand in hands)


class SwipeDetector:
    def __init__(self):
        self.history = deque()  # (시각, 손목 x, 손목 y)

    def reset(self):
        self.history.clear()

    def update(self, now, landmarks):
        x, y = landmarks[WRIST].x, landmarks[WRIST].y
        self.history.append((now, x, y))
        while now - self.history[0][0] > config.SWIPE_WINDOW_SEC:
            self.history.popleft()

        moved = x - self.history[0][1]
        moved_y = y - self.history[0][2]
        if abs(moved) < hand_size(landmarks) * config.SWIPE_RATIO:
            return None
        if abs(moved) < abs(moved_y):  # 옆보다 위아래로 더 움직였으면 손을 내리는 동작이다
            return None
        # 카메라는 나를 마주 보므로, 내가 오른쪽으로 움직이면 영상의 x 는 작아진다
        return "SWIPE_RIGHT" if moved < 0 else "SWIPE_LEFT"


def get_command(hands, swipe, now):
    if not hands:  # 빠르게 움직이면 손을 잠깐 놓친다 → 쓸기 기록은 지우지 않는다
        return None
    if len(hands) == 2:  # 양손이 보이면 명령으로 보지 않는다 (시작/종료 동작과 구분)
        swipe.reset()
        return None

    hand = hands[0]
    if is_ok(hand.landmarks):
        return "OK"
    return swipe.update(now, hand.landmarks)

"""IDLE/ACTIVE 상태 전환과 오작동 필터(연속 프레임 + 쿨다운)."""
import config
from gestures import both_hands


class RemoteState:
    def __init__(self):
        self.mode = "IDLE"
        self.hold_since = None
        self.last_seen = 0.0
        self.last_command = None
        self.streak = 0
        self.cool_until = 0.0

    def update_mode(self, now, hands):
        if hands:
            self.last_seen = now

        if self.mode == "ACTIVE" and now - self.last_seen > config.NO_HAND_SEC:
            self._switch("IDLE")
            return

        wanted = "Open_Palm" if self.mode == "IDLE" else "Closed_Fist"
        if not both_hands(hands, wanted):
            self.hold_since = None
            return

        if self.hold_since is None:
            self.hold_since = now
        if now - self.hold_since >= config.HOLD_SEC:
            self._switch("ACTIVE" if self.mode == "IDLE" else "IDLE")

    def should_fire(self, command, now):
        if command != self.last_command:
            self.last_command = command
            self.streak = 0
        if command is None:
            return False

        self.streak += 1
        # 쓸기는 이미 일정 시간의 이동량으로 판정했으므로 첫 프레임에 바로 인정한다
        needed = config.STABLE_FRAMES if command == "OK" else 1
        # == 로 비교: OK 를 계속 들고 있어도 키는 한 번만 나간다 (손을 풀어야 다시 가능)
        if self.streak == needed and now >= self.cool_until:
            self.cool_until = now + config.COOLDOWN_SEC
            return True
        return False

    def _switch(self, mode):
        self.mode = mode
        self.hold_since = None
        self.last_command = None
        self.streak = 0

질문 대비

설계 문서 4절의 공부 범위를 기준으로 정리했습니다. 답을 보지 않고 자기 말로 말해 보세요.

A. main.py · recognizer.py

GestureRecognizerOptions 인자는?model_asset_path 모델 파일, running_mode 입력 방식, num_hands 최대 손 개수, min_hand_detection_confidence 검출 기준, min_tracking_confidence 추적 기준입니다.
RunningMode 3종의 차이와 VIDEO를 고른 이유는?IMAGE는 사진 한 장, VIDEO는 연속 프레임을 넣고 결과를 바로 받기, LIVE_STREAM은 결과를 콜백으로 나중에 받기입니다. 웹캠 루프를 위에서 아래로 단순하게 읽을 수 있어 VIDEO를 골랐습니다.
BGR을 RGB로 바꾸는 이유는?OpenCV 프레임은 BGR 순서이고, mp.Image에 SRGB로 넘기려면 RGB 순서여야 합니다.
recognize_for_video에 timestamp_ms가 필요한 이유는?VIDEO 모드는 프레임 순서를 알아야 이전 프레임을 이용해 추적할 수 있습니다. 시각은 계속 커져야 합니다.

B. gestures.py

GestureRecognizerResult 4개 필드는?gestures, handedness, hand_landmarks, hand_world_landmarks입니다.
정규화 좌표와 world 좌표의 차이는?정규화 좌표는 이미지 가로·세로를 0~1로 본 위치이고, world 좌표는 손 중심 기준의 실제 크기(미터)입니다. 이 코드는 정규화 좌표를 씁니다.
score로 거르는 이유는?모델이 확신하지 못한 제스처까지 믿으면 오작동합니다. MIN_SCORE보다 낮으면 None(터미널 표시는 제스처 아님)으로 처리합니다.
OK는 어떻게 판정하나요?4번과 8번의 거리가 손 크기 × 0.3보다 작고, 중지·약지·새끼 끝이 관절보다 손목에서 멀면 OK입니다.
쓸기는 어떻게 판정하나요?최근 0.8초 사이 손목 x 이동량이 손 크기보다 크고 위아래 이동보다 크면 쓸기입니다. 카메라는 사용자를 마주 보므로 사용자가 오른쪽으로 움직이면 x가 작아집니다.

C. state.py · controller.py · config.py

IDLE/ACTIVE로 나눈 이유는?IDLE에서는 어떤 제스처도 키를 누르지 않아, 평소 손동작이 명령이 되는 걸 막습니다.
연속 프레임과 쿨다운은 각각 무엇을 막나요?연속 프레임은 한두 프레임 잘못 나온 OK를 막고, 쿨다운은 쓸고 되돌아오는 손 같은 직후 입력을 막습니다.
키가 유튜브로 가는 원리는?pynput은 OS에 키 입력을 만들어 줄 뿐이고, OS가 포커스된 창으로 보냅니다. 그래서 유튜브를 클릭해 둬야 합니다.
macOS에서 손쉬운 사용 권한이 필요한 이유는?다른 앱으로 키 입력을 보내는 기능이라 OS가 사용자 허가를 요구합니다.
왜 값을 config.py에 모았나요?테스트하며 자주 바꾸는 값을 한 파일에서 고치기 위해서입니다. 제스처-키 매핑도 여기만 바꾸면 PPT나 음악 플레이어에 쓸 수 있습니다.
